Solution: N, S = map(int, input().split())

YES = False

for x in range(N - 1 + 1, S - (N - 1)):
    K = x
    if S - K > N - 1 and S - K < S - (N - 1):
        YES = True
        break

if YES:
    print('YES')
    ans = ''
    for _ in range(N-1):
        ans += '1 '
    ans += str(S - (N - 1))
    print(ans)
    print(K)
else:
    print('NO')

            a.append('1')
            b.append('1')

            a.append('0')
            b.append('0')
            
        if c[i] == '1' and flag == 1:
            a.append('0')
            b.append('1')
            
        if c[i] == '1' and  flag == 0:
            a.append('1')
            b.append('0')
            flag = 1
    a1 = ''.join(a)
    b1 = ''.join(b)
    print(a1)
    print(b1)
    a.clear()
    b.clear()
    t -= 1

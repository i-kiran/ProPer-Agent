input = lambda:sys.stdin.readline().strip()
def solve():
    n,m = map(int,input().split())
    a = []
    for i in range(n):
        a.append(int(input(),2))
    need = ((1<<m)-n-1)//2 + 1
    cur = (1<<(m-1))-1
    while True:
        left = cur+1
        flag = False 
        for i in range(n):
            flag|=a[i]==cur 
            if a[i]<=cur:
                left-=1
        if left==need and not flag:
            ans = bin(cur)[2:]
            print('0'*(m-len(ans))+ans)
            return
        elif left<need:
            cur+=1
        else:
            cur-=1
t = int(input())
while t:
    t-=1
    solve()
